\section{State Space and the Kaprekar Map}
\label{sec:prelim}

Let $X$ be the set of all non-decreasing 4-tuples of decimal digits that are not all equal:

The size of $X$ is $|X| = 705$.

The Kaprekar map $K: X\to X$ is defined by
\[

\section{Theorem Chain}
\label{sec:chain}

\subsection{A1: Algebraic Identity}

\begin{theorem}[Algebraic Identity]\label{thm:A1}
For any $(a,b,c,d)\in X$,
\[
D(a,b,c,d) := \text{desc} - \text{asc} = 999\,(d-a) + 90\,(c-b) = 999\,g_1 + 90\,g_2.
\]
\end{theorem}
\begin{proof}
Direct expansion:
\begin{align*}
D &= (1000d+100c+10b+a) - (1000a+100b+10c+d) \\
  &= 1000(d-a) + 100(c-b) + 10(b-c) + (a-d) \\
  &= 1000(d-a) + 100(c-b) - 10(c-b) - (d-a) \\
  &= (1000-1)(d-a) + (100-10)(c-b) = 999g_1 + 90g_2.
\end{align*}
\end{proof}
Thus $D$ depends only on the gap vector $g=(g_1,g_2)$.

\subsection{A2*: Strong Deterministic Lifting}

Let $S: \mathbb{Z}_{\ge0} \to X$ be the sorting function that sends a non-negative integer $D<10000$ to the sorted quadruple of its decimal digits (with leading zeros). Because $D(s)$ is a function of $g$ alone, we obtain:

\begin{theorem}[Strong Deterministic Lifting]\label{thm:A2star}
There exists a unique map $F: G\to X$ such that
\[
K(s) = F(\pi(s)) \qquad \forall s\in X.
\]
Explicitly, $F(g) = S(999\,g_1+90\,g_2)$.
\end{theorem}
\begin{proof}
By Theorem~\ref{thm:A1}, $D(s)=999g_1+90g_2$ depends only on $g=\pi(s)$. Hence $K(s)=S(D(s))$ depends only on $g$.
\end{proof}

$\pi(K(s))$ depends only on $\pi(s)$.

\subsection{A3: Semiconjugacy}

\begin{definition}

\[
\widetilde K(g) = \pi(F(g)).
\]

\begin{theorem}[Semiconjugacy]\label{thm:A3}
The following diagram commutes:
\[
\pi \circ K = \widetilde K \circ \pi \quad \text{on } X.
\]
\end{theorem}
\begin{proof}
For any $s\in X$, let $g=\pi(s)$. Then $\pi(K(s)) = \pi(F(g)) = \widetilde K(g) = \widetilde K(\pi(s))$.
\end{proof}

\subsection{A4: Bisimulation Stability}

Let $P_0 = X/\pi$ be the partition of $X$ into gap fibers. The refinement operator $R$ splits a block if its elements map to different blocks.

\begin{theorem}[Bisimulation Stability]\label{thm:A4}
$R(P_0) = P_0$. That is, $P_0$ is a fixed point of partition refinement.
\end{theorem}
\begin{proof}
For any $s_1,s_2$ with $\pi(s_1)=\pi(s_2)=g$, Theorem~\ref{thm:A3} gives $\pi(K(s_1)) = \widetilde K(g) = \pi(K(s_2))$. Thus all elements of $\pi^{-1}(g)$ map into the same block $\pi^{-1}(\widetilde K(g))$, so no block is split.
\end{proof}

\subsection{GS: Gap Sufficiency}

The semiconjugacy extends to all iterates.

\begin{theorem}[Gap Sufficiency]\label{thm:GS}
If $\pi(s_1)=\pi(s_2)$, then for every $n\ge 0$,
\[
\pi(K^n(s_1)) = \pi(K^n(s_2)).
\]
Thus the entire future gap trajectory is determined by the initial gap state.
\end{theorem}
\begin{proof}

\pi(K^{n+1}(s_1)) = \widetilde K(\pi(K^n(s_1))) = \widetilde K(g) = \widetilde K(\pi(K^n(s_2))) = \pi(K^{n+1}(s_2)).
\]
\end{proof}

\section{Gap Realization Theorem}
\label{sec:gaprealization}

The gap space $G$ was defined as the image of $\pi$ on the finite set $X$. We now characterize $G$ combinatorially.

\begin{theorem}[Gap Realization]
\[
G = \{ (g_1,g_2) \in \mathbb{Z}^2 : 0 \le g_2 \le g_1 \le 9,\; g_1 \ge 1 \}.
\]
Consequently $|G| = 54$.
\end{theorem}
\begin{proof}
\textit{Necessity:} For any $(a,b,c,d)\in X$, the sorted order implies $d\ge c\ge b\ge a$, so $g_1 = d-a \ge 0$ and $g_2 = c-b \ge 0$. Moreover $d-a \ge c-b$ because $d-c + b-a \ge 0$, hence $g_1 \ge g_2$. Also $g_1=0$ would force $a=d$, which with sortedness implies all digits equal, contradicting the definition of $X$. Thus $g_1\ge 1$. All digits lie in $\{0,\dots,9\}$, so $g_1 \le 9$ and $g_2 \le g_1 \le 9$.

\textit{Sufficiency:} For any pair $(g_1,g_2)$ satisfying $0\le g_2\le g_1\le 9$, $g_1\ge1$, we construct a sorted quadruple in $X$ with that gap. If $g_1+g_2\le 9$, take $(a,b,c,d) = (0,0,g_2, g_1+g_2)$. This is sorted and $d-a=g_1$, $c-b=g_2$. If $g_1+g_2 > 9$, set $a = 9-g_1$, $b = 9-g_1$, $c = 9-g_1+g_2$, $d = 9$. Since $g_2\le g_1$, we have $c\le 9$, and all digits are between $0$ and $9$. The tuple is sorted and satisfies $d-a = 9 - (9-g_1)=g_1$, $c-b = (9-g_1+g_2) - (9-g_1)=g_2$. Finally, $g_1\ge1$ guarantees the tuple is not a repdigit.

\textit{Counting:} For each $g_1 = 1,\dots,9$, the admissible $g_2$ values are $0,1,\dots,g_1$, giving $g_1+1$ possibilities. Hence
\[
|G| = \sum_{g_1=1}^{9} (g_1+1) = \sum_{k=2}^{10} k = 54.
\]
\end{proof}

\section{Introduction}

\section{Definitions}

\begin{definition}[State Spaces]
Let $X_{\text{raw}} = {n \in \mathbb{Z} : 1000 \leq n \leq 9999,\ \text{not all digits equal}}$ be the raw 4-digit strings ($|X_{\text{raw}}| = 9990$). Let $X = {(a,b,c,d) \in \mathbb{Z}^4 : 0 \leq a \leq b \leq c \leq d \leq 9,\ \text{not all equal}}$ be the sorted non-repdigit states ($|X| = 705$).

\begin{definition}[Kaprekar Map]
For $s = (a,b,c,d) \in X$, define $K: X \to X$ by $K(s) = \text{sort}(\text{desc}(s) - \text{asc}(s))$ where $\text{desc}(s) = 1000d + 100c + 10b + a$ and $\text{asc}(s) = 1000a + 100b + 10c + d$.
\end{definition}

\begin{definition}[Gap Projection]
Define $\pi: X \to G$ by $\pi(a,b,c,d) = (d-a, c-b)$ where $G = \pi(X) \subset \mathbb{Z}^2$.
\end{definition}

\section{The Algebraic Core}

\begin{lemma}[A1: Algebraic Identity]\label{lem:a1}
For any $s = (a,b,c,d) \in X$ with $g_1 = d-a$ and $g_2 = c-b$:
\begin{equation}
\text{desc}(s) - \text{asc}(s) = 999g_1 + 90g_2.
\end{equation}
\end{lemma}

\begin{proof}
Expand:
\begin{align}
\text{desc}(s) - \text{asc}(s) &= (1000d + 100c + 10b + a) - (1000a + 100b + 10c + d) \
&= 999(d-a) + 90(c-b) \
&= 999g_1 + 90g_2. \qedhere
\end{align}
\end{proof}

\begin{lemma}[A2*: Deterministic Lifting]\label{lem:a2star}
There exists a uniquely defined map $F: G \to X$ such that for all $s \in X$:
\begin{equation}
K(s) = F(\pi(s)).
\end{equation}
\end{lemma}

\begin{proof}
By Lemma \ref{lem:a1}, the difference $D = \text{desc}(s) - \text{asc}(s) = 999g_1 + 90g_2$ depends only on $g = \pi(s)$. The next state $K(s)$ is the sorted digit state of $D$. Since $D$ depends only on $g$, and sorting is a deterministic function, $K(s)$ depends only on $g$. Define $F(g) = S(999g_1 + 90g_2)$ where $S$ extracts and sorts decimal digits. Then $K(s) = F(\pi(s))$.
\end{proof}

\begin{lemma}[A2: Observable Determinism]\label{lem:a2}
For all $s \in X$, $\pi(K(s))$ depends only on $\pi(s)$.
\end{lemma}

\begin{proof}
Immediate from Lemma \ref{lem:a2star} by composing with $\pi$: $\pi(K(s)) = \pi(F(\pi(s)))$.
\end{proof}

\begin{lemma}[A3: Semiconjugacy]\label{lem:a3}
Define $\widetilde{K}: G \to G$ by $\widetilde{K}(g) = \pi(F(g))$. Then:
\begin{equation}
\pi \circ K = \widetilde{K} \circ \pi.
\end{equation}
\end{lemma}

\begin{proof}
For any $s \in X$: $\pi(K(s)) = \pi(F(\pi(s))) = \widetilde{K}(\pi(s))$ by definition of $\widetilde{K}$.
\end{proof}

\begin{theorem}[A4: Bisimulation Stability]\label{thm:a4}
The gap partition $P_0 = X / \pi$ is a fixed point of the refinement operator $\mathcal{R}$. That is, $\mathcal{R}(P_0) = P_0$.
\end{theorem}

\begin{proof}
For any block $B_g = \pi^{-1}(g) \in P_0$ and any $x, y \in B_g$: by Lemma \ref{lem:a3}, $\pi(K(x)) = \widetilde{K}(g) = \pi(K(y))$. Thus $K(x)$ and $K(y)$ lie in the same block $B_{\widetilde{K}(g)}$. The refinement operator does not split $B_g$. Since this holds for all $g \in G$, $\mathcal{R}(P_0) = P_0$.
\end{proof}

\section{The Gap Space}

\begin{theorem}[Gap Realization]\label{thm:gr}
The gap state space is exactly:
\begin{equation}
G = {(g_1, g_2) \in \mathbb{Z}^2 : 0 \leq g_2 \leq g_1 \leq 9,\ g_1 \geq 1}.
\end{equation}
Therefore $|G| = 54$.
\end{theorem}

\begin{proof}
$(\subseteq)$ For any $\pi(a,b,c,d) = (g_1, g_2)$: $g_1 = d-a \geq 0$, $g_2 = c-b \geq 0$, and $g_1 - g_2 = (d-c) + (b-a) \geq 0$ implies $g_1 \geq g_2$. Also $g_1 = d-a \leq 9$ and non-repdigit implies $g_1 \geq 1$.

$(\supseteq)$ For any $(g_1, g_2)$ in the region, the state $(0, 0, g_2, g_1)$ satisfies $0 \leq 0 \leq g_2 \leq g_1$, is non-repdigit (since $g_1 \geq 1$), and has gap $(g_1, g_2)$.
